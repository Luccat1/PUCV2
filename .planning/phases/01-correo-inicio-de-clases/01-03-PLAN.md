  <script>
    var niveles = [];

    function showState(name) {
      ['loading','form','preview','result'].forEach(function(s) {
        document.getElementById('state-' + s).style.display = (s === name) ? '' : 'none';
      });
    }

    function showForm() {
      showState('form');
    }

    // On load: fetch active levels from server
    window.onload = function() {
      google.script.run
        .withSuccessHandler(function(result) {
          niveles = result;
          if (!niveles || niveles.length === 0) {
            showState('result');
            var box = document.getElementById('result-text');
            box.textContent = 'No hay estudiantes pendientes de notificación de inicio de clases en Lista Final Curso.';
            box.className = 'result-box result-ok';
            return;
          }
          var container = document.getElementById('sala-fields');
          container.innerHTML = '';
          niveles.forEach(function(n) {
            var lbl = document.createElement('label');
            lbl.textContent = 'Sala — ' + n;
            lbl.setAttribute('for', 'sala-' + n);
            var inp = document.createElement('input');
            inp.type = 'text';
            inp.id = 'sala-' + n;
            inp.name = n;
            inp.placeholder = 'Ej: Sala 101';
            inp.autocomplete = 'off';
            container.appendChild(lbl);
            container.appendChild(inp);
          });
          showState('form');
        })
        .withFailureHandler(function(err) {
          showState('result');
          var box = document.getElementById('result-text');
          box.textContent = 'Error al cargar niveles: ' + err.message;
          box.className = 'result-box result-err';
        })
        .getNivelesActivos();
    };

    function submitSalas() {
      var salas = {};
      var allFilled = true;
      niveles.forEach(function(n) {
        var val = document.getElementById('sala-' + n).value.trim();
        if (!val) allFilled = false;
        salas[n] = val;
      });
      if (!allFilled) {
        alert('Por favor ingresa sala para todos los niveles antes de continuar.');
        return;
      }

      document.getElementById('btn-preview').disabled = true;
      document.getElementById('form-spinner').style.display = '';

      google.script.run
        .withSuccessHandler(function(preview) {
          document.getElementById('btn-preview').disabled = false;
          document.getElementById('form-spinner').style.display = 'none';
          document.getElementById('preview-text').textContent = preview;
          showState('preview');
        })
        .withFailureHandler(function(err) {
          document.getElementById('btn-preview').disabled = false;
          document.getElementById('form-spinner').style.display = 'none';
          alert('Error: ' + err.message);
        })
        .guardarSalasYObtenerPreview(salas);
    }

    function confirmarEnvio() {
      document.getElementById('btn-confirm').disabled = true;
      document.getElementById('confirm-spinner').style.display = '';

      google.script.run
        .withSuccessHandler(function(msg) {
          document.getElementById('btn-confirm').disabled = false;
          document.getElementById('confirm-spinner').style.display = 'none';
          showState('result');
          var box = document.getElementById('result-text');
          box.textContent = msg;
          box.className = 'result-box ' + (msg.startsWith('ERROR') ? 'result-err' : 'result-ok');
        })
        .withFailureHandler(function(err) {
          document.getElementById('btn-confirm').disabled = false;
          document.getElementById('confirm-spinner').style.display = 'none';
          showState('result');
          var box = document.getElementById('result-text');
          box.textContent = 'Error al enviar: ' + err.message;
          box.className = 'result-box result-err';
        })
        .enviarCorreosInicioClases();
    }
  </script>
</body>
</html>
